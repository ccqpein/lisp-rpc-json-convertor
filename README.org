#+TITLE: Lisp-RPC JSON Convertor
#+AUTHOR: ccQpein

=lisp-rpc-json-convertor= is a CLI tool and Rust library for converting between Lisp-RPC data structures and JSON bidirectionally.

* Features

- Convert named Lisp-RPC structures into tagged JSON objects and vice versa.
- Convert anonymous maps into standard JSON objects and vice versa.
- Convert Lisp-RPC lists into JSON arrays and vice versa.
- Convert scalar types (strings, numbers, booleans, null/nil).
- Support for single data records, multi-record files, and stdin streams.
- CLI application with =-d/--data=, =-j/--json=, =-f/--file=, auto-detection, and customizable =-k/--key=.
- Rust library API with extension traits for both Lisp-RPC =Data= and =serde_json::Value=.

* Data Format Mapping

| Lisp-RPC                                      | JSON                                                                        |
|-----------------------------------------------+-----------------------------------------------------------------------------|
| =(book-info :title "1984" :id 1 :available t)= | ={"data_name": "book-info", "title": "1984", "id": 1, "available": true}=  |
| ='(:first "John" :last "Doe")=                  | ={"first": "John", "last": "Doe"}=                                          |
| ='("a" "b" "c")=                                | =["a", "b", "c"]=                                                           |
| ="hello"= / =42= / =3.14=                           | ="hello"= / =42= / =3.14=                                                       |
| =t= / =nil= / =false=                           | =true= / =null= / =false=                                                   |

* CLI Usage

** Installation / Build

#+begin_src bash
  cargo install lisp-rpc-json-convertor
#+end_src

** Options

| Flag                  | Description                                            |
|-----------------------+--------------------------------------------------------|
| =-d, --data <DATA>=     | Pass raw Lisp-RPC string directly (to JSON)            |
| =-j, --json <JSON>=     | Pass raw JSON string directly (to Lisp-RPC)            |
| =-f, --file <FILE>=     | Read from a file (or =-= for stdin)                      |
| =-o, --output <OUTPUT>= | Write output to a file (defaults to stdout)            |
| =-k, --key <KEY>=       | Custom tag key name (defaults to ="data_name"=)          |
| =-p, --pretty=          | Pretty-print JSON with indentation (when to JSON)      |
| =[INPUT]=               | Positional raw string or file path (auto-detects mode) |

** Examples

*** Convert JSON to Lisp-RPC
#+begin_src shell
  lisp-rpc-json-convertor -j '{"data_name": "book", "title": "Dune", "year": 1965}'
  # (book :title "Dune" :year 1965)
#+end_src

*** Convert JSON with custom tag key
#+begin_src shell
  lisp-rpc-json-convertor -j '{"_type": "user", "name": "Alice"}' -k "_type"
  # (user :name "Alice")
#+end_src

*** Convert JSON array of objects to Lisp-RPC list
#+begin_src shell
  lisp-rpc-json-convertor -j '[{"data_name": "book", "id": 1}, {"data_name": "book", "id": 2}]'
  # '((book :id 1) (book :id 2))
#+end_src

*** Convert Lisp-RPC to JSON
#+begin_src shell
  lisp-rpc-json-convertor -d '(book :title "Dune" :year 1965)' --pretty
  # {
  #   "data_name": "book",
  #   "title": "Dune",
  #   "year": 1965
  # }
#+end_src

*** Convert files
#+begin_src shell
  # JSON file to Lisp-RPC
  lisp-rpc-json-convertor -f data.json -o data.lisp

  # Lisp-RPC file to JSON
  lisp-rpc-json-convertor -f input.lisp -o output.json --pretty
#+end_src

*** Pipe via stdin
#+begin_src shell
  echo '{"data_name": "get-book", "title": "Dune"}' | lisp-rpc-json-convertor
  # (get-book :title "Dune")

  echo '(get-book :title "Dune")' | lisp-rpc-json-convertor --pretty
  # { "data_name": "get-book", "title": "Dune" }
#+end_src

#+begin_src toml
  [dependencies]

#+end_src

** JSON to Lisp-RPC Conversion

#+begin_src rust
  use lisp_rpc_json_convertor::{from_json_str, from_json_str_to_string, JsonLispRpcExt};
  use serde_json::json;

  fn main() -> anyhow::Result<()> {
      // Direct string conversion
      let lisp_str = from_json_str_to_string(r#"{"data_name": "book", "title": "1984"}"#)?;
      println!("{}", lisp_str); // (book :title "1984")

      // Using extension trait on serde_json::Value
      let val = json!({
          "data_name": "server",
          "host": "localhost",
          "port": 8080
      });
      let data = val.to_lisp_rpc()?;
      println!("{}", data.to_string()); // (server :host "localhost" :port 8080)
      Ok(())
  }
#+end_src

** Lisp-RPC to JSON Conversion

#+begin_src rust
  use lisp_rpc_json_convertor::{to_json_from_str, LispRpcJsonExt};
  use lisp_rpc_rust_raw_data::Data;

  fn main() -> anyhow::Result<()> {
      let json_val = to_json_from_str(r#"(book :title "1984" :id 1)"#)?;
      println!("{}", json_val);

      // Using extension trait on Data
      let data = Data::from_root_str(r#"(server :host "localhost" :port 8080)"#, None)?;
      let val = data.to_json()?;
      println!("{}", val);
      Ok(())
  }
#+end_src

** Custom Convertor Configuration

#+begin_src rust
  use lisp_rpc_json_convertor::Convertor;
  use serde_json::json;

  fn main() -> anyhow::Result<()> {
      let convertor = Convertor::new().data_name_key("_type");

      // JSON to Lisp-RPC with custom key
      let json_val = json!({ "_type": "user-info", "name": "Alice" });
      let data = convertor.from_json(&json_val)?;
      println!("{}", data.to_string()); // (user-info :name "Alice")

      // Lisp-RPC to JSON with custom key
      let back_json = convertor.to_json(&data)?;
      println!("{}", back_json); // {"_type":"user-info","name":"Alice"}
      Ok(())
  }
#+end_src
